chapa("========================= \n");
chapa("       NAYA BANK      \n");
chapa("=========================\n");


kaam deposit(balance, amount) {
    firta balance + amount;
}

kaam withdraw(balance, amount) {
    yedi amount > balance {
        chapa "Withdrawal denied!\n";
        firta balance;
    }

    firta balance - amount;
}

kaam interest(balance, rate, years) {
    manam i := 0;

    jabasamma i < years {
        balance := balance + balance * rate / 100;
        i := i + 1;
    }

    firta balance;
}


manam balance := 1000;
chapa "Initial balance: $";
chapa balance;

balance := kaam_gar(deposit(balance, 500));
balance := kaam_gar(withdraw(balance, 300));
balance := kaam_gar(interest(balance, 5, 3));

chapa "\nFinal balance: $";
chapa balance;
chapa "\n=====================\n";
